  # Exercise the example bodies without discovering or sending input to real hosts.
  # Only the example's host constant and loop duration differ in these test copies.
  foreach(SD_EXAMPLE MouseMove Touch SendKeyboard ReciveImgPacket ReciveAudio)
    set(SD_EXAMPLE_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/samples/${SD_EXAMPLE}/main.cpp")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${SD_EXAMPLE_SOURCE}")
    file(READ "${SD_EXAMPLE_SOURCE}" SD_EXAMPLE_CODE)
    string(REPLACE "../common/" "${CMAKE_CURRENT_SOURCE_DIR}/samples/common/" SD_EXAMPLE_CODE "${SD_EXAMPLE_CODE}")
    string(REPLACE [=[sample::find_host(L"", L"", 28252)]=]
      [=[sample::Host{"127.0.0.1", uint16_t(std::stoi(std::getenv("SD_TEST_PORT")))}]=]
      SD_EXAMPLE_CODE "${SD_EXAMPLE_CODE}")
    if(SD_EXAMPLE STREQUAL "Touch")
      string(REPLACE "loop(10000)" "loop(1200)" SD_EXAMPLE_CODE "${SD_EXAMPLE_CODE}")
    else()
      string(REPLACE "loop(10000)" "loop(200)" SD_EXAMPLE_CODE "${SD_EXAMPLE_CODE}")
    endif()
    set(SD_EXAMPLE_COPY "${CMAKE_CURRENT_BINARY_DIR}/sample-tests/${SD_EXAMPLE}.cpp")
    file(WRITE "${SD_EXAMPLE_COPY}" "#include <cstdlib>\n${SD_EXAMPLE_CODE}")
    add_executable(sd_test_${SD_EXAMPLE} "${SD_EXAMPLE_COPY}")
    target_link_libraries(sd_test_${SD_EXAMPLE} PRIVATE spacedesk_sdk)
    target_compile_definitions(sd_test_${SD_EXAMPLE} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  endforeach()
